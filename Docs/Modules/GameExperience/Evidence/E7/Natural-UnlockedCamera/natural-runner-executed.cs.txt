    public static void Capture(string name)
    {
        var view = EditorWindow.GetWindow(typeof(EditorWindow).Assembly.GetType("UnityEditor.GameView")); view.Repaint();
        var render = (RenderTexture)view.GetType().GetField("m_RenderTexture", Private).GetValue(view);
        var old = RenderTexture.active; var image = new Texture2D(render.width, render.height, TextureFormat.RGB24, false);
        try
        {
            RenderTexture.active = render; image.ReadPixels(new Rect(0, 0, render.width, render.height), 0, 0); image.Apply();
            if (SystemInfo.graphicsUVStartsAtTop)
            {
                var pixels = image.GetPixels32();
                for (int y = 0; y < render.height / 2; y++) for (int x = 0; x < render.width; x++)
                { int a = y * render.width + x, b = (render.height - 1 - y) * render.width + x; var p = pixels[a]; pixels[a] = pixels[b]; pixels[b] = p; }
                image.SetPixels32(pixels); image.Apply();
            }
            Directory.CreateDirectory(Dir); File.WriteAllBytes(Dir + name, image.EncodeToPNG());
        }
        finally { RenderTexture.active = old; UnityEngine.Object.DestroyImmediate(image); }
    }

                if (gp.IsMatchOver && ++endingFrames >= 4)
                {
                    Balance();
                    File.WriteAllText(Dir + "natural-result.json", Newtonsoft.Json.JsonConvert.SerializeObject(new
                    { outcome = gp.Outcome.ToString(), completedNights = root.Reports.CompletedNight, signals = root.DawnReport.DawnSignalCount,
                        total = root.Reports.Total, wallet = gp.Wallet, resultText = Find<MatchSession>().ResultStatisticsText, duration = elapsed,
                        wallSeconds = EditorApplication.timeSinceStartup - wall, unity = Application.unityVersion, editor = true, timeScaleBeforeTerminal = 1,
                        durations = new[] { clock.Definition.DayDuration, clock.Definition.DuskDuration, clock.Definition.NightDuration }, fogEnabled = !fog.IsDisabled }, Newtonsoft.Json.Formatting.Indented));
                    GameExperienceE7Validation.Capture("natural-result.png"); Finish("");
                }
            }
            catch (Exception error) { Finish(error.ToString()); }
        };
        EditorApplication.update += update;
        EditorApplication.isPaused = false;
        return "Armed natural five-night Editor run (675 game seconds, watchdog1100wall). No manual simulation/clock ticks; writing progress every15game seconds.";
    }
